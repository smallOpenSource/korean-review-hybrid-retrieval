  conn.close()

  korean = 0
  for r in rows:

      if has_korean:
          korean += 1
          print(f'✅ {r[\"review_id\"][:8]}: {emotions}')
      else:
          print(f'⚠️  {r[\"review_id\"][:8]}: {emotions}')

  print('✅ 정상' if korean >= 5 else '❌ 실패')
